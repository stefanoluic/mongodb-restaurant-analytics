# Upiti v1


### q01 - Koji gradovi imaju najjacu ponudu visoko ocenjenih vegetarijanskih ili veganskih restorana?

Uloga: Turisticka organizacija

Kolekcija: `restaurant_classifications`

```js
db.restaurant_classifications.aggregate([
    {
      $match: {
        $or: [
          {
            vegetarian_friendly: true
          },
          {
            vegan_options: true
          }
        ]
      }
    },
    {
      $lookup: {
        from: "restaurant_ratings",
        localField: "_id",
        foreignField: "_id",
        as: "rating"
      }
    },
    {
      $unwind: "$rating"
    },
    {
      $match: {
        "rating.avg_rating": {
          $gte: 4.2
        },
        "rating.total_reviews_count": {
          $gte: 50
        }
      }
    },
    {
      $lookup: {
        from: "restaurant_locations",
        localField: "_id",
        foreignField: "_id",
        as: "location"
      }
    },
    {
      $unwind: "$location"
    },
    {
      $match: {
        "location.country": {
          $ne: null
        },
        "location.city": {
          $ne: null
        }
      }
    },
    {
      $group: {
        _id: {
          country: "$location.country",
          city: "$location.city"
        },
        restaurant_count: {
          $sum: 1
        },
        avg_rating: {
          $avg: "$rating.avg_rating"
        },
        avg_reviews: {
          $avg: "$rating.total_reviews_count"
        }
      }
    },
    {
      $match: {
        restaurant_count: {
          $gte: 10
        }
      }
    },
    {
      $sort: {
        restaurant_count: -1,
        avg_rating: -1
      }
    },
    {
      $limit: 20
    },
    {
      $project: {
        _id: 0,
        country: "$_id.country",
        city: "$_id.city",
        restaurant_count: 1,
        avg_rating: {
          $round: [
            "$avg_rating",
            2
          ]
        },
        avg_reviews: {
          $round: [
            "$avg_reviews",
            1
          ]
        }
      }
    }
  ])
```

Kolekcija: `restaurant_classifications`

```js

    {
      $match: {
        meals: {
          $all: [
            "Lunch",

          ]
        }
      }
    },
    {
      $lookup: {
        from: "restaurants",
        localField: "_id",
        foreignField: "_id",
        as: "restaurant"
      }
    },
    {
      $unwind: "$restaurant"
    },
    {
      $match: {
        "restaurant.price_score": {
          $in: [
            1,
            2
          ]
        }
      }
    },
    {
      $lookup: {
        from: "restaurant_ratings",
        localField: "_id",
        foreignField: "_id",
        as: "rating"
      }
    },
    {
      $unwind: "$rating"
    },
    {
      $match: {
        "rating.avg_rating": {
          $gte: 4.0
        },
        "rating.total_reviews_count": {
          $gte: 30
        },
        "rating.category_ratings.value": {
          $ne: null
        }
      }
    },
    {
      $lookup: {
        from: "restaurant_locations",
        localField: "_id",
        foreignField: "_id",
        as: "location"

      $match: {
        "location.country": {
          $ne: null

          $ne: null

        value_for_money: {
```

```js

    {
      $match: {
```
```js
        avg_rating: {
```
```js
          $gte: 4.2
        },
```

```js
          $gte: 100
        }
      }
    },
    {
      $lookup: {
        from: "restaurants",
        localField: "_id",
        foreignField: "_id",
        as: "restaurant"
      }
    },
    {
      $unwind: "$restaurant"
    },
    {
      $lookup: {
        from: "restaurant_locations",
        localField: "_id",
        foreignField: "_id",
        as: "location"
      }
    },
    {
      $unwind: "$location"
    },
    {
      $match: {
        "location.country": {
          $ne: null
        },
        "location.city": {
          $ne: null
        }
      }
    },
    {
      $group: {
        _id: {
          country: "$location.country",
          city: "$location.city"
        },
        quality_demand_count: {
          $sum: 1
        },
        known_price_count: {
          $sum: {
            $cond: [
              {
                $ne: [
                  "$restaurant.price_score",
                  null
                ]
              },
              1,
              0
            ]
          }
        },
        premium_restaurant_count: {
          $sum: {
            $cond: [
              {
                $eq: [
                  "$restaurant.price_score",
                  3
                ]
              },
              1,
              0
            ]
          }
        },
        avg_rating: {
          $avg: "$avg_rating"
        },
        avg_reviews: {
          $avg: "$total_reviews_count"
        }
      }
    },
    {
      $addFields: {
        premium_share: {
          $cond: [
            {
              $gt: [
                "$known_price_count",
                0
              ]
            },
            {
              $divide: [
                "$premium_restaurant_count",
                "$known_price_count"
              ]
            },
            null
          ]
        }

          }
```
